cli_py = r'''
"""
visual_verifier.cli

V5.1a package CLI wrapper.

This module delegates to the already-working V5.0b single-file CLI.

Usage:
    python -m visual_verifier.cli batch --cases <cases.csv> --output <output_dir> --run

Reason for this bridge:
    V5.0b proved the CLI engine works.
    V5.1a creates a package entrypoint without breaking the working engine.
    V5.1b will move the engine logic into package modules.
"""

import runpy
import sys
from pathlib import Path


def main():
    package_file = Path(__file__).resolve()
    project_root = package_file.parents[1]

    legacy_cli = (
        project_root
        / "visual_verifier_cli"
        / "visual_verify_cli_v5_0b.py"
    )

    if not legacy_cli.exists():
        print("ERROR: Legacy V5.0b CLI script was not found.")
        print(f"Expected path: {legacy_cli}")
        return 1

    original_argv = sys.argv[:]

    sys.argv = [str(legacy_cli)] + sys.argv[1:]

    try:
        runpy.run_path(str(legacy_cli), run_name="__main__")
    except SystemExit as exc:
        if isinstance(exc.code, int):
            return exc.code
        if exc.code is None:
            return 0
        return 1
    finally:
        sys.argv = original_argv

    return 0


if __name__ == "__main__":
    sys.exit(main())
'''

io_utils_py = r'''
"""
I/O utility helpers.

V5.1a placeholder:
    CSV, JSON, folder, and path helpers will be moved here in V5.1b.
"""

import json
from pathlib import Path


def ensure_dir(path):
    path = Path(path)
    path.mkdir(parents=True, exist_ok=True)
    return path


def write_json(data, path):
    path = Path(path)
    ensure_dir(path.parent)

    with path.open("w", encoding="utf-8") as f:
        json.dump(data, f, indent=4, ensure_ascii=False, default=str)

    return path
