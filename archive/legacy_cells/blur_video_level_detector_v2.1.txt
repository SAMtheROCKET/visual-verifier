def get_video_info(video_path):
    cap = cv2.VideoCapture(video_path)

    if not cap.isOpened():
        raise FileNotFoundError(f"Could not open video: {video_path}")

    fps = cap.get(cv2.CAP_PROP_FPS)
    frame_count = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
    width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))

    duration_sec = frame_count / fps if fps and fps > 0 else 0.0

    cap.release()

    return {
        "path": video_path,
        "fps": float(fps),
        "frame_count": int(frame_count),
        "width": int(width),
        "height": int(height),
        "duration_sec": float(duration_sec)
    }

def safe_crop(image, x1, y1, x2, y2):
    h, w = image.shape[:2]

    x1 = max(0, min(w, int(x1)))
    y1 = max(0, min(h, int(y1)))
    x2 = max(0, min(w, int(x2)))
    y2 = max(0, min(h, int(y2)))

    return image[y1:y2, x1:x2]


def calculate_iou(box_a, box_b):
    ax1, ay1, ax2, ay2 = box_a
    bx1, by1, bx2, by2 = box_b

    inter_x1 = max(ax1, bx1)
    inter_y1 = max(ay1, by1)
    inter_x2 = min(ax2, bx2)
    inter_y2 = min(ay2, by2)

    inter_w = max(0, inter_x2 - inter_x1)
    inter_h = max(0, inter_y2 - inter_y1)
    inter_area = inter_w * inter_h

    area_a = max(0, ax2 - ax1) * max(0, ay2 - ay1)
    area_b = max(0, bx2 - bx1) * max(0, by2 - by1)

    union_area = area_a + area_b - inter_area

    if union_area <= 0:
        return 0.0

    return inter_area / union_area


def max_consecutive_gap(missing_frames):
    if not missing_frames:
        return 0

    max_gap = 1
    current_gap = 1

    for i in range(1, len(missing_frames)):
        if missing_frames[i] == missing_frames[i - 1] + 1:
            current_gap += 1
            max_gap = max(max_gap, current_gap)
        else:
            current_gap = 1

    return max_gap


def convert_for_json(obj):
    """
    Robust converter for JSON export.
    Handles Python, NumPy, Pandas, bool, NaN, inf, list, tuple, set, dict.
    """

    if obj is None:
        return None

    if isinstance(obj, (bool, np.bool_)):
        return bool(obj)

    if isinstance(obj, dict):
        return {str(k): convert_for_json(v) for k, v in obj.items()}

    if isinstance(obj, list):
        return [convert_for_json(v) for v in obj]

    if isinstance(obj, tuple):
        return [convert_for_json(v) for v in obj]

    if isinstance(obj, set):
        return [convert_for_json(v) for v in obj]

    if isinstance(obj, np.ndarray):
        return [convert_for_json(v) for v in obj.tolist()]

    if isinstance(obj, np.integer):
        return int(obj)

    if isinstance(obj, np.floating):
        value = float(obj)

        if np.isinf(value):
            return "inf"

        if np.isnan(value):
            return None

        return value

    if isinstance(obj, float):
        if np.isinf(obj):
            return "inf"

        if np.isnan(obj):
            return None

        return obj

    try:
        if pd.isna(obj):
            return None
    except Exception:
        pass

    return obj

def calculate_laplacian_variance(image_crop):
    if image_crop is None or image_crop.size == 0:
        return 0.0

    gray = cv2.cvtColor(image_crop, cv2.COLOR_BGR2GRAY)
    laplacian_var = cv2.Laplacian(gray, cv2.CV_64F).var()

    return float(laplacian_var)


def calculate_severity_score(
    mean_diff,
    changed_ratio,
    laplacian_raw,
    laplacian_processed
):
    diff_component = min(1.0, mean_diff / DIFF_NORMALIZER)

    density_component = min(
        1.0,
        changed_ratio / CHANGED_RATIO_NORMALIZER
    )

    epsilon = 1e-9

    if laplacian_raw <= epsilon:
        edge_change_ratio = 0.0
    else:
        edge_change_ratio = abs(laplacian_processed - laplacian_raw) / (laplacian_raw + epsilon)

    edge_component = min(1.0, edge_change_ratio)

    severity_score = (
        0.45 * diff_component +
        0.25 * density_component +
        0.30 * edge_component
    ) * 100.0

    severity_score = round(float(severity_score), 2)
    edge_change_ratio = round(float(edge_change_ratio), 6)

    return severity_score, edge_change_ratio


def get_severity_label(severity_score):
    if severity_score >= HIGH_THRESHOLD:
        return "VERY HIGH"
    elif severity_score >= MEDIUM_THRESHOLD:
        return "HIGH"
    elif severity_score >= LOW_THRESHOLD:
        return "MEDIUM"
    elif severity_score >= VERY_LOW_THRESHOLD:
        return "LOW"
    else:
        return "VERY LOW"


def calculate_blur_percentage(laplacian_raw, laplacian_processed):
    epsilon = 1e-9

    if laplacian_raw <= epsilon:
        return float("inf")

    return float(laplacian_processed * 100.0 / laplacian_raw)

def evaluate_region_candidate(region):
    rejection_reasons = []

    if region["changed_ratio"] < MIN_CHANGED_RATIO_FOR_ACCEPTED_REGION:
        rejection_reasons.append("changed_ratio_too_low")

    if region["mean_diff"] < MIN_MEAN_DIFF_FOR_ACCEPTED_REGION:
        rejection_reasons.append("mean_diff_too_low")

    if region["severity_score"] < MIN_SEVERITY_SCORE_FOR_ACCEPTED_REGION:
        rejection_reasons.append("severity_score_too_low")

    if len(rejection_reasons) > 0:
        return False, rejection_reasons

    return True, []

def detect_multiple_processed_regions(raw_frame, processed_frame):
    if raw_frame.shape != processed_frame.shape:
        processed_frame = cv2.resize(
            processed_frame,
            (raw_frame.shape[1], raw_frame.shape[0])
        )

    gray_raw = cv2.cvtColor(raw_frame, cv2.COLOR_BGR2GRAY)
    gray_processed = cv2.cvtColor(processed_frame, cv2.COLOR_BGR2GRAY)

    diff = cv2.absdiff(gray_raw, gray_processed)

    _, threshold_mask = cv2.threshold(
        diff,
        DIFF_THRESHOLD,
        255,
        cv2.THRESH_BINARY
    )

    kernel = cv2.getStructuringElement(
        cv2.MORPH_RECT,
        (MERGE_KERNEL_SIZE, MERGE_KERNEL_SIZE)
    )

    cleaned_mask = cv2.morphologyEx(
        threshold_mask,
        cv2.MORPH_CLOSE,
        kernel
    )

    cleaned_mask = cv2.dilate(
        cleaned_mask,
        kernel,
        iterations=1
    )

    contours, _ = cv2.findContours(
        cleaned_mask,
        cv2.RETR_EXTERNAL,
        cv2.CHAIN_APPROX_SIMPLE
    )

    frame_h, frame_w = raw_frame.shape[:2]

    accepted_boxes = []
    rejected_boxes = []

    candidate_id = 0

    for contour in contours:
        x, y, w, h = cv2.boundingRect(contour)

        x1 = max(0, x - BOX_PADDING)
        y1 = max(0, y - BOX_PADDING)
        x2 = min(frame_w, x + w + BOX_PADDING)
        y2 = min(frame_h, y + h + BOX_PADDING)

        box_width = x2 - x1
        box_height = y2 - y1
        box_area = box_width * box_height

        if box_area < MIN_BOX_AREA:
            continue

        if box_width < MIN_BOX_WIDTH or box_height < MIN_BOX_HEIGHT:
            continue

        candidate_id += 1

        diff_crop = diff[y1:y2, x1:x2]
        threshold_crop = threshold_mask[y1:y2, x1:x2]

        changed_pixels = int(np.sum(threshold_crop > 0))
        changed_ratio = changed_pixels / box_area if box_area > 0 else 0.0

        mean_diff = float(np.mean(diff_crop))
        max_diff = float(np.max(diff_crop))

        raw_crop = safe_crop(raw_frame, x1, y1, x2, y2)
        processed_crop = safe_crop(processed_frame, x1, y1, x2, y2)

        laplacian_raw = calculate_laplacian_variance(raw_crop)
        laplacian_processed = calculate_laplacian_variance(processed_crop)

        blur_percentage = calculate_blur_percentage(
            laplacian_raw,
            laplacian_processed
        )

        severity_score, edge_change_ratio = calculate_severity_score(
            mean_diff=mean_diff,
            changed_ratio=changed_ratio,
            laplacian_raw=laplacian_raw,
            laplacian_processed=laplacian_processed
        )

        severity_label = get_severity_label(severity_score)

        region = {
            "candidate_id": int(candidate_id),
            "x1": int(x1),
            "y1": int(y1),
            "x2": int(x2),
            "y2": int(y2),
            "width": int(box_width),
            "height": int(box_height),
            "area": int(box_area),
            "changed_pixels": int(changed_pixels),
            "changed_ratio": float(changed_ratio),
            "mean_diff": float(mean_diff),
            "max_diff": float(max_diff),
            "laplacian_raw": float(laplacian_raw),
            "laplacian_processed": float(laplacian_processed),
            "blur_percentage_processed_vs_raw": float(blur_percentage),
            "edge_change_ratio": float(edge_change_ratio),
            "severity_score": float(severity_score),
            "severity_label": severity_label,
            "track_id": -1,
            "track_iou": 0.0,
            "region_filter_status": "ACCEPTED",
            "rejection_reasons": []
        }

        accepted, rejection_reasons = evaluate_region_candidate(region)

        if ENABLE_REGION_FALSE_POSITIVE_FILTER and not accepted:
            region["region_filter_status"] = "REJECTED"
            region["rejection_reasons"] = rejection_reasons
            rejected_boxes.append(region)
        else:
            accepted_boxes.append(region)

    accepted_boxes = sorted(accepted_boxes, key=lambda b: (b["y1"], b["x1"]))
    rejected_boxes = sorted(rejected_boxes, key=lambda b: (b["y1"], b["x1"]))

    return accepted_boxes, rejected_boxes, diff, threshold_mask, cleaned_mask

class SimpleIoUTracker:
    def __init__(self, iou_threshold=0.20, max_missing_frames=3):
        self.iou_threshold = iou_threshold
        self.max_missing_frames = max_missing_frames
        self.next_track_id = 1
        self.tracks = {}

    def update(self, boxes, frame_number):
        assigned_detection_indices = set()

        for track_id in list(self.tracks.keys()):
            track = self.tracks[track_id]
            old_bbox = track["bbox"]

            best_iou = 0.0
            best_det_idx = None

            for det_idx, box in enumerate(boxes):
                if det_idx in assigned_detection_indices:
                    continue

                new_bbox = (
                    box["x1"],
                    box["y1"],
                    box["x2"],
                    box["y2"]
                )

                iou = calculate_iou(old_bbox, new_bbox)

                if iou > best_iou:
                    best_iou = iou
                    best_det_idx = det_idx

            if best_det_idx is not None and best_iou >= self.iou_threshold:
                boxes[best_det_idx]["track_id"] = int(track_id)
                boxes[best_det_idx]["track_iou"] = round(float(best_iou), 6)

                self.tracks[track_id]["bbox"] = (
                    boxes[best_det_idx]["x1"],
                    boxes[best_det_idx]["y1"],
                    boxes[best_det_idx]["x2"],
                    boxes[best_det_idx]["y2"]
                )

                self.tracks[track_id]["last_frame"] = int(frame_number)
                self.tracks[track_id]["missing_count"] = 0

                assigned_detection_indices.add(best_det_idx)

            else:
                self.tracks[track_id]["missing_count"] += 1

        for det_idx, box in enumerate(boxes):
            if det_idx in assigned_detection_indices:
                continue

            track_id = self.next_track_id
            self.next_track_id += 1

            box["track_id"] = int(track_id)
            box["track_iou"] = 1.0

            self.tracks[track_id] = {
                "bbox": (
                    box["x1"],
                    box["y1"],
                    box["x2"],
                    box["y2"]
                ),
                "first_frame": int(frame_number),
                "last_frame": int(frame_number),
                "missing_count": 0
            }

        for track_id in list(self.tracks.keys()):
            if self.tracks[track_id]["missing_count"] > self.max_missing_frames:
                del self.tracks[track_id]

        return boxes

def draw_boxes_on_frame(frame, boxes, frame_number, total_frames, frame_status):
    annotated = frame.copy()

    for box_id, box in enumerate(boxes, start=1):
        x1 = box["x1"]
        y1 = box["y1"]
        x2 = box["x2"]
        y2 = box["y2"]

        cv2.rectangle(
            annotated,
            (x1, y1),
            (x2, y2),
            BOX_COLOR,
            2
        )

        label = (
            f"T{box['track_id']} "
            f"B{box_id} "
            f"S:{box['severity_score']} "
            f"{box['severity_label']}"
        )

        cv2.putText(
            annotated,
            label,
            (x1, max(20, y1 - 8)),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.52,
            BOX_COLOR,
            2,
            cv2.LINE_AA
        )

    header_color = FAIL_COLOR if "FAIL" in frame_status else TEXT_COLOR

    if len(boxes) > 0:
        max_severity = max(box["severity_score"] for box in boxes)
        track_ids = sorted(list(set([box["track_id"] for box in boxes])))
    else:
        max_severity = 0.0
        track_ids = []

    header_text = (
        f"Frame {frame_number}/{total_frames} | "
        f"Regions: {len(boxes)} | "
        f"Tracks: {track_ids} | "
        f"MaxS: {round(max_severity, 2)} | "
        f"{frame_status}"
    )

    cv2.putText(
        annotated,
        header_text,
        (20, 35),
        cv2.FONT_HERSHEY_SIMPLEX,
        0.62,
        header_color,
        2,
        cv2.LINE_AA
    )

    return annotated

def get_row_as_dict(row):
    if row is None:
        return None

    result = {}

    for key, value in row.items():
        result[key] = convert_for_json(value)

    return result


def build_warnings(raw_info, processed_info):
    warnings = []

    if raw_info["frame_count"] != processed_info["frame_count"]:
        warnings.append(
            "Raw and processed videos have different frame counts. Only common frames were checked."
        )

    if raw_info["fps"] != processed_info["fps"]:
        warnings.append(
            "Raw and processed videos have different FPS metadata. V2.1 compares frames by index, not timestamp alignment."
        )

    if raw_info["width"] != processed_info["width"] or raw_info["height"] != processed_info["height"]:
        warnings.append(
            "Raw and processed videos have different resolutions. Processed frames were resized to match raw frames."
        )

    if PROCESS_EVERY_N_FRAMES != 1:
        warnings.append(
            f"Sampling is enabled: only every {PROCESS_EVERY_N_FRAMES} frame(s) were checked. This is not recommended for safety-critical verification."
        )

    if MAX_FRAMES is not None:
        warnings.append(
            f"MAX_FRAMES is set to {MAX_FRAMES}. Only a limited part of the video was checked."
        )

    if not EXPECT_PROCESSING_IN_EVERY_FRAME:
        warnings.append(
            "EXPECT_PROCESSING_IN_EVERY_FRAME is False. Frames with no processing are not treated as failures."
        )

    if ENABLE_REGION_FALSE_POSITIVE_FILTER:
        warnings.append(
            "Region false-positive filtering is enabled. Very weak candidate regions may be rejected before tracking."
        )

    if ENABLE_TRACK_CLEANUP:
        warnings.append(
            "Track cleanup is enabled. Very short and weak tracks are marked as probable false positives."
        )

    return warnings


def calculate_track_confidence_score(
    frames_present_count,
    presence_ratio,
    max_severity_score,
    max_area_px
):
    duration_component = min(1.0, frames_present_count / 5.0)
    presence_component = min(1.0, presence_ratio)
    severity_component = min(1.0, max_severity_score / 60.0)
    area_component = min(1.0, max_area_px / 2500.0)

    confidence_score = (
        0.25 * duration_component +
        0.25 * presence_component +
        0.30 * severity_component +
        0.20 * area_component
    ) * 100.0

    return round(float(confidence_score), 2)


def get_track_confidence_label(track_confidence_score):
    if track_confidence_score >= 75:
        return "HIGH"
    elif track_confidence_score >= 50:
        return "MEDIUM"
    elif track_confidence_score >= 35:
        return "LOW"
    else:
        return "VERY LOW"


def build_track_report(df_region_report):
    if df_region_report is None or len(df_region_report) == 0:
        return pd.DataFrame([])

    track_rows = []

    for track_id, group in df_region_report.groupby("track_id"):
        group = group.sort_values(by="frame_number")

        frames_present = sorted(group["frame_number"].unique().astype(int).tolist())
        first_frame = int(min(frames_present))
        last_frame = int(max(frames_present))

        expected_frames = list(range(first_frame, last_frame + 1))
        missing_frames = sorted(list(set(expected_frames) - set(frames_present)))

        track_duration_frames = len(expected_frames)
        frames_present_count = len(frames_present)
        missing_frame_count = len(missing_frames)

        presence_ratio = (
            frames_present_count / track_duration_frames
            if track_duration_frames > 0 else 0.0
        )

        avg_severity_score = round(float(group["severity_score"].mean()), 2)
        max_severity_score = round(float(group["severity_score"].max()), 2)
        min_severity_score = round(float(group["severity_score"].min()), 2)

        avg_area_px = round(float(group["area_px"].mean()), 2)
        max_area_px = round(float(group["area_px"].max()), 2)

        track_confidence_score = calculate_track_confidence_score(
            frames_present_count=frames_present_count,
            presence_ratio=presence_ratio,
            max_severity_score=max_severity_score,
            max_area_px=max_area_px
        )

        track_confidence_label = get_track_confidence_label(track_confidence_score)

        track_status = "CONTINUOUS"

        if missing_frame_count > 0:
            track_status = "HAS_TEMPORAL_GAPS"

        is_single_frame_track = frames_present_count == 1

        probable_false_positive_reasons = []

        if is_single_frame_track:
            probable_false_positive_reasons.append("single_frame_track")

        if max_severity_score < WEAK_TRACK_MAX_SEVERITY_SCORE:
            probable_false_positive_reasons.append("weak_max_severity")

        if track_confidence_score < LOW_CONFIDENCE_TRACK_THRESHOLD:
            probable_false_positive_reasons.append("low_track_confidence")

        is_probable_false_positive_track = False

        if ENABLE_TRACK_CLEANUP:
            if (
                frames_present_count <= SHORT_TRACK_MAX_PRESENT_FRAMES
                and max_severity_score < WEAK_TRACK_MAX_SEVERITY_SCORE
            ):
                is_probable_false_positive_track = True

            if track_confidence_score < LOW_CONFIDENCE_TRACK_THRESHOLD:
                is_probable_false_positive_track = True

        if is_probable_false_positive_track:
            clean_track_status = "PROBABLE_FALSE_POSITIVE_TRACK"
        else:
            clean_track_status = "CLEAN_TRACK"

        track_rows.append({
            "track_id": int(track_id),
            "track_status": track_status,
            "clean_track_status": clean_track_status,
            "is_probable_false_positive_track": bool(is_probable_false_positive_track),
            "probable_false_positive_reasons": probable_false_positive_reasons,
            "track_confidence_score": float(track_confidence_score),
            "track_confidence_label": track_confidence_label,
            "first_frame": int(first_frame),
            "last_frame": int(last_frame),
            "track_duration_frames": int(track_duration_frames),
            "frames_present_count": int(frames_present_count),
            "missing_frame_count": int(missing_frame_count),
            "presence_ratio": round(float(presence_ratio), 6),
            "frames_present": frames_present,
            "missing_frames": missing_frames,
            "max_consecutive_missing_frames": int(max_consecutive_gap(missing_frames)),
            "total_regions_in_track": int(len(group)),
            "avg_severity_score": float(avg_severity_score),
            "max_severity_score": float(max_severity_score),
            "min_severity_score": float(min_severity_score),
            "avg_area_px": float(avg_area_px),
            "max_area_px": float(max_area_px),
            "avg_changed_ratio": round(float(group["changed_ratio"].mean()), 6),
            "avg_mean_diff": round(float(group["mean_diff"].mean()), 4)
        })

    return pd.DataFrame(track_rows)

def verify_video_level_v2_1(raw_video_path, processed_video_path):
    raw_info = get_video_info(raw_video_path)
    processed_info = get_video_info(processed_video_path)

    warnings = build_warnings(raw_info, processed_info)

    print("====================================================")
    print("RAW VIDEO INFO")
    print("====================================================")
    print(raw_info)

    print("\n====================================================")
    print("PROCESSED VIDEO INFO")
    print("====================================================")
    print(processed_info)

    if len(warnings) > 0:
        print("\n====================================================")
        print("WARNINGS")
        print("====================================================")

        for warning in warnings:
            print("-", warning)

    raw_cap = cv2.VideoCapture(raw_video_path)
    processed_cap = cv2.VideoCapture(processed_video_path)

    raw_fps = raw_info["fps"]
    raw_width = raw_info["width"]
    raw_height = raw_info["height"]

    if raw_fps is None or raw_fps <= 0:
        raw_fps = 30
        warnings.append("Raw FPS could not be read correctly. Fallback FPS = 30 was used.")

    raw_frame_count = raw_info["frame_count"]
    processed_frame_count = processed_info["frame_count"]

    total_common_frames = min(raw_frame_count, processed_frame_count)

    if MAX_FRAMES is not None:
        total_common_frames = min(total_common_frames, MAX_FRAMES)

    fourcc = cv2.VideoWriter_fourcc(*"mp4v")

    video_writer = cv2.VideoWriter(
        OUTPUT_ANNOTATED_VIDEO,
        fourcc,
        raw_fps,
        (raw_width, raw_height)
    )

    tracker = SimpleIoUTracker(
        iou_threshold=TRACK_IOU_THRESHOLD,
        max_missing_frames=MAX_TRACK_MISSING_FRAMES
    )

    frame_report_rows = []
    region_report_rows = []
    rejected_region_report_rows = []

    failed_frames = []
    frames_checked = 0
    frames_skipped = 0
    frames_with_processing = 0
    frames_without_processing = 0
    total_detected_regions = 0
    total_candidate_regions = 0
    total_rejected_regions = 0
    global_region_id = 0
    global_rejected_region_id = 0

    print("\n====================================================")
    print("STARTING V2.1 FRAME-BY-FRAME VERIFICATION + TRACKING + CLEANUP")
    print("====================================================")

    frame_index = 0

    while frame_index < total_common_frames:
        ret_raw, raw_frame = raw_cap.read()
        ret_processed, processed_frame = processed_cap.read()

        if not ret_raw or not ret_processed:
            break

        frame_number = frame_index + 1
        should_process = (frame_index % PROCESS_EVERY_N_FRAMES == 0)

        if not should_process:
            frames_skipped += 1
            frame_index += 1
            continue

        if processed_frame.shape != raw_frame.shape:
            processed_frame = cv2.resize(
                processed_frame,
                (raw_frame.shape[1], raw_frame.shape[0])
            )

        boxes, rejected_boxes, diff, threshold_mask, cleaned_mask = detect_multiple_processed_regions(
            raw_frame,
            processed_frame
        )

        boxes = tracker.update(boxes, frame_number)

        num_regions = len(boxes)
        num_rejected_regions = len(rejected_boxes)
        num_candidate_regions = num_regions + num_rejected_regions

        total_candidate_regions += num_candidate_regions
        total_rejected_regions += num_rejected_regions
        total_detected_regions += num_regions

        if num_regions > 0:
            frames_with_processing += 1
            frame_status = "PASS - PROCESSING DETECTED"
        else:
            frames_without_processing += 1

            if EXPECT_PROCESSING_IN_EVERY_FRAME:
                frame_status = "FAIL - NO PROCESSING DETECTED"
                failed_frames.append(frame_number)
            else:
                frame_status = "NO PROCESSING DETECTED"

        annotated_frame = draw_boxes_on_frame(
            processed_frame,
            boxes,
            frame_number,
            total_common_frames,
            frame_status
        )

        video_writer.write(annotated_frame)

        timestamp_sec = round(frame_index / raw_fps, 4) if raw_fps > 0 else None

        if SAVE_EXTRACTED_FRAMES:
            raw_frame_path = os.path.join(
                RAW_FRAMES_DIR,
                f"raw_frame_{frame_number:06d}.png"
            )

            processed_frame_path = os.path.join(
                PROCESSED_FRAMES_DIR,
                f"{PROCESSED_VIDEO_FILE_NAME}_processed_frame_{frame_number:06d}.png"
            )

            annotated_frame_path = os.path.join(
                ANNOTATED_FRAMES_DIR,
                f"{PROCESSED_VIDEO_FILE_NAME}_{OUTPUT_VERSION}_annotated_frame_{frame_number:06d}.png"
            )

            cv2.imwrite(raw_frame_path, raw_frame)
            cv2.imwrite(processed_frame_path, processed_frame)
            cv2.imwrite(annotated_frame_path, annotated_frame)

        if num_regions > 0:
            total_changed_pixels = sum(box["changed_pixels"] for box in boxes)
            total_box_area = sum(box["area"] for box in boxes)
            avg_mean_diff = float(np.mean([box["mean_diff"] for box in boxes]))
            max_diff = float(np.max([box["max_diff"] for box in boxes]))
            largest_region_area = max([box["area"] for box in boxes])
            avg_severity_score = float(np.mean([box["severity_score"] for box in boxes]))
            max_severity_score = float(np.max([box["severity_score"] for box in boxes]))
            detected_track_ids = sorted(list(set([int(box["track_id"]) for box in boxes])))
        else:
            total_changed_pixels = 0
            total_box_area = 0
            avg_mean_diff = 0.0
            max_diff = 0.0
            largest_region_area = 0
            avg_severity_score = 0.0
            max_severity_score = 0.0
            detected_track_ids = []

        frame_area = raw_width * raw_height
        processed_area_ratio = total_box_area / frame_area if frame_area > 0 else 0.0

        frame_report_rows.append({
            "processed_video_file_name": PROCESSED_VIDEO_FILE_NAME,
            "frame_number": int(frame_number),
            "timestamp_sec": timestamp_sec,
            "num_candidate_regions": int(num_candidate_regions),
            "num_rejected_candidate_regions": int(num_rejected_regions),
            "num_detected_regions": int(num_regions),
            "detected_track_ids": detected_track_ids,
            "total_region_area_px": int(total_box_area),
            "processed_area_ratio": round(float(processed_area_ratio), 6),
            "total_changed_pixels": int(total_changed_pixels),
            "avg_mean_diff": round(float(avg_mean_diff), 4),
            "max_diff": round(float(max_diff), 4),
            "largest_region_area_px": int(largest_region_area),
            "avg_severity_score": round(float(avg_severity_score), 2),
            "max_severity_score": round(float(max_severity_score), 2),
            "frame_status": frame_status
        })

        for box_id, box in enumerate(boxes, start=1):
            global_region_id += 1

            x1 = box["x1"]
            y1 = box["y1"]
            x2 = box["x2"]
            y2 = box["y2"]

            raw_crop = safe_crop(raw_frame, x1, y1, x2, y2)
            processed_crop = safe_crop(processed_frame, x1, y1, x2, y2)
            annotated_crop = safe_crop(annotated_frame, x1, y1, x2, y2)

            raw_crop_path = ""
            processed_crop_path = ""
            annotated_crop_path = ""

            if SAVE_REGION_CROPS:
                region_name = (
                    f"{PROCESSED_VIDEO_FILE_NAME}"
                    f"_{OUTPUT_VERSION}"
                    f"_frame_{frame_number:06d}"
                    f"_track_{box['track_id']:03d}"
                    f"_box_{box_id:03d}"
                )

                raw_crop_path = os.path.join(
                    REGION_CROPS_DIR,
                    f"{region_name}_raw.png"
                )

                processed_crop_path = os.path.join(
                    REGION_CROPS_DIR,
                    f"{region_name}_processed.png"
                )

                annotated_crop_path = os.path.join(
                    REGION_CROPS_DIR,
                    f"{region_name}_annotated.png"
                )

                cv2.imwrite(raw_crop_path, raw_crop)
                cv2.imwrite(processed_crop_path, processed_crop)
                cv2.imwrite(annotated_crop_path, annotated_crop)

            blur_percentage = box["blur_percentage_processed_vs_raw"]

            if np.isfinite(blur_percentage):
                blur_percentage_value = round(float(blur_percentage), 4)
            else:
                blur_percentage_value = "inf"

            region_report_rows.append({
                "processed_video_file_name": PROCESSED_VIDEO_FILE_NAME,
                "global_region_id": int(global_region_id),
                "track_id": int(box["track_id"]),
                "track_iou": round(float(box["track_iou"]), 6),
                "frame_number": int(frame_number),
                "timestamp_sec": timestamp_sec,
                "box_id_in_frame": int(box_id),
                "x1": int(x1),
                "y1": int(y1),
                "x2": int(x2),
                "y2": int(y2),
                "width": int(box["width"]),
                "height": int(box["height"]),
                "area_px": int(box["area"]),
                "changed_pixels": int(box["changed_pixels"]),
                "changed_ratio": round(float(box["changed_ratio"]), 6),
                "mean_diff": round(float(box["mean_diff"]), 4),
                "max_diff": round(float(box["max_diff"]), 4),
                "laplacian_raw": round(float(box["laplacian_raw"]), 4),
                "laplacian_processed": round(float(box["laplacian_processed"]), 4),
                "blur_percentage_processed_vs_raw": blur_percentage_value,
                "edge_change_ratio": float(box["edge_change_ratio"]),
                "severity_score": float(box["severity_score"]),
                "severity_label": box["severity_label"],
                "region_filter_status": box["region_filter_status"],
                "raw_crop_path": raw_crop_path,
                "processed_crop_path": processed_crop_path,
                "annotated_crop_path": annotated_crop_path
            })

        for rejected_box_id, rejected_box in enumerate(rejected_boxes, start=1):
            global_rejected_region_id += 1

            x1 = rejected_box["x1"]
            y1 = rejected_box["y1"]
            x2 = rejected_box["x2"]
            y2 = rejected_box["y2"]

            raw_crop = safe_crop(raw_frame, x1, y1, x2, y2)
            processed_crop = safe_crop(processed_frame, x1, y1, x2, y2)

            rejected_raw_crop_path = ""
            rejected_processed_crop_path = ""

            if SAVE_REJECTED_REGION_CROPS:
                rejected_region_name = (
                    f"{PROCESSED_VIDEO_FILE_NAME}"
                    f"_{OUTPUT_VERSION}"
                    f"_frame_{frame_number:06d}"
                    f"_rejected_{rejected_box_id:03d}"
                )

                rejected_raw_crop_path = os.path.join(
                    REJECTED_REGION_CROPS_DIR,
                    f"{rejected_region_name}_raw.png"
                )

                rejected_processed_crop_path = os.path.join(
                    REJECTED_REGION_CROPS_DIR,
                    f"{rejected_region_name}_processed.png"
                )

                cv2.imwrite(rejected_raw_crop_path, raw_crop)
                cv2.imwrite(rejected_processed_crop_path, processed_crop)

            rejected_region_report_rows.append({
                "processed_video_file_name": PROCESSED_VIDEO_FILE_NAME,
                "global_rejected_region_id": int(global_rejected_region_id),
                "frame_number": int(frame_number),
                "timestamp_sec": timestamp_sec,
                "rejected_box_id_in_frame": int(rejected_box_id),
                "x1": int(x1),
                "y1": int(y1),
                "x2": int(x2),
                "y2": int(y2),
                "width": int(rejected_box["width"]),
                "height": int(rejected_box["height"]),
                "area_px": int(rejected_box["area"]),
                "changed_pixels": int(rejected_box["changed_pixels"]),
                "changed_ratio": round(float(rejected_box["changed_ratio"]), 6),
                "mean_diff": round(float(rejected_box["mean_diff"]), 4),
                "max_diff": round(float(rejected_box["max_diff"]), 4),
                "severity_score": float(rejected_box["severity_score"]),
                "severity_label": rejected_box["severity_label"],
                "rejection_reasons": rejected_box["rejection_reasons"],
                "raw_crop_path": rejected_raw_crop_path,
                "processed_crop_path": rejected_processed_crop_path
            })

        frames_checked += 1

        print(
            f"Frame {frame_number}/{total_common_frames} | "
            f"Candidates: {num_candidate_regions} | "
            f"Accepted: {num_regions} | "
            f"Rejected: {num_rejected_regions} | "
            f"Tracks: {detected_track_ids} | "
            f"Max severity: {round(max_severity_score, 2)} | "
            f"Status: {frame_status}"
        )

        frame_index += 1

    raw_cap.release()
    processed_cap.release()
    video_writer.release()

    df_frame_report = pd.DataFrame(frame_report_rows)
    df_region_report = pd.DataFrame(region_report_rows)
    df_rejected_region_report = pd.DataFrame(rejected_region_report_rows)
    df_track_report = build_track_report(df_region_report)

    df_frame_report.to_csv(
        OUTPUT_FRAME_REPORT,
        index=False,
        encoding="utf-8-sig"
    )

    df_region_report.to_csv(
        OUTPUT_REGION_REPORT,
        index=False,
        encoding="utf-8-sig"
    )

    df_rejected_region_report.to_csv(
        OUTPUT_REJECTED_REGION_REPORT,
        index=False,
        encoding="utf-8-sig"
    )

    df_track_report.to_csv(
        OUTPUT_TRACK_REPORT,
        index=False,
        encoding="utf-8-sig"
    )

    #============================================================
    # SUMMARY CALCULATION
    #============================================================

    if frames_checked > 0:
        processing_coverage_percent = frames_with_processing * 100.0 / frames_checked
        failed_frame_percent = len(failed_frames) * 100.0 / frames_checked
        avg_regions_per_checked_frame = total_detected_regions / frames_checked

        if total_candidate_regions > 0:
            rejection_rate_percent = total_rejected_regions * 100.0 / total_candidate_regions
        else:
            rejection_rate_percent = 0.0
    else:
        processing_coverage_percent = 0.0
        failed_frame_percent = 0.0
        avg_regions_per_checked_frame = 0.0
        rejection_rate_percent = 0.0

    if len(df_frame_report) > 0:
        strongest_frame_row = df_frame_report.sort_values(
            by="max_severity_score",
            ascending=False
        ).iloc[0]

        strongest_frame = get_row_as_dict(strongest_frame_row)

        processed_only_df = df_frame_report[df_frame_report["num_detected_regions"] > 0]

        if len(processed_only_df) > 0:
            weakest_processed_frame_row = processed_only_df.sort_values(
                by="max_severity_score",
                ascending=True
            ).iloc[0]

            weakest_processed_frame = get_row_as_dict(weakest_processed_frame_row)
        else:
            weakest_processed_frame = None
    else:
        strongest_frame = None
        weakest_processed_frame = None

    if len(df_region_report) > 0:
        avg_region_area = float(df_region_report["area_px"].mean())
        max_region_area = float(df_region_report["area_px"].max())
        avg_changed_ratio = float(df_region_report["changed_ratio"].mean())
        avg_region_mean_diff = float(df_region_report["mean_diff"].mean())
        avg_severity = float(df_region_report["severity_score"].mean())
        max_severity = float(df_region_report["severity_score"].max())

        severity_counts_raw = df_region_report["severity_label"].value_counts().to_dict()

        severity_counts = {
            "VERY LOW": int(severity_counts_raw.get("VERY LOW", 0)),
            "LOW": int(severity_counts_raw.get("LOW", 0)),
            "MEDIUM": int(severity_counts_raw.get("MEDIUM", 0)),
            "HIGH": int(severity_counts_raw.get("HIGH", 0)),
            "VERY HIGH": int(severity_counts_raw.get("VERY HIGH", 0))
        }

        highest_severity_region_row = df_region_report.sort_values(
            by="severity_score",
            ascending=False
        ).iloc[0]

        highest_severity_region = get_row_as_dict(highest_severity_region_row)

        strongest_region_text = (
            f"Frame {int(highest_severity_region_row['frame_number'])}, "
            f"Track {int(highest_severity_region_row['track_id'])}, "
            f"Box {int(highest_severity_region_row['box_id_in_frame'])}, "
            f"Severity {highest_severity_region_row['severity_score']} "
            f"({highest_severity_region_row['severity_label']})"
        )

        unique_tracks_count = int(df_region_report["track_id"].nunique())

    else:
        avg_region_area = 0.0
        max_region_area = 0.0
        avg_changed_ratio = 0.0
        avg_region_mean_diff = 0.0
        avg_severity = 0.0
        max_severity = 0.0

        severity_counts = {
            "VERY LOW": 0,
            "LOW": 0,
            "MEDIUM": 0,
            "HIGH": 0,
            "VERY HIGH": 0
        }

        highest_severity_region = None
        strongest_region_text = "No region detected"
        unique_tracks_count = 0

    if len(df_track_report) > 0:
        tracks_with_gaps_df = df_track_report[df_track_report["missing_frame_count"] > 0]
        tracks_with_gaps_count = int(len(tracks_with_gaps_df))
        continuous_tracks_count = int(len(df_track_report) - tracks_with_gaps_count)

        probable_false_positive_tracks_df = df_track_report[
            df_track_report["is_probable_false_positive_track"] == True
        ]

        clean_tracks_df = df_track_report[
            df_track_report["is_probable_false_positive_track"] == False
        ]

        probable_false_positive_tracks_count = int(len(probable_false_positive_tracks_df))
        clean_tracks_count = int(len(clean_tracks_df))

        clean_tracks_with_gaps_df = clean_tracks_df[
            clean_tracks_df["missing_frame_count"] > 0
        ]

        clean_tracks_with_gaps_count = int(len(clean_tracks_with_gaps_df))

        all_track_missing_frames = sorted(
            list(
                set(
                    frame
                    for missing_list in df_track_report["missing_frames"].tolist()
                    for frame in missing_list
                )
            )
        )

        clean_track_gap_frames = sorted(
            list(
                set(
                    frame
                    for missing_list in clean_tracks_df["missing_frames"].tolist()
                    for frame in missing_list
                )
            )
        )

        if len(tracks_with_gaps_df) > 0:
            worst_gap_track_row = tracks_with_gaps_df.sort_values(
                by=["missing_frame_count", "max_consecutive_missing_frames"],
                ascending=False
            ).iloc[0]

            worst_gap_track = get_row_as_dict(worst_gap_track_row)
        else:
            worst_gap_track = None

        if len(clean_tracks_with_gaps_df) > 0:
            worst_clean_gap_track_row = clean_tracks_with_gaps_df.sort_values(
                by=["missing_frame_count", "max_consecutive_missing_frames"],
                ascending=False
            ).iloc[0]

            worst_clean_gap_track = get_row_as_dict(worst_clean_gap_track_row)
        else:
            worst_clean_gap_track = None

    else:
        tracks_with_gaps_count = 0
        continuous_tracks_count = 0
        probable_false_positive_tracks_count = 0
        clean_tracks_count = 0
        clean_tracks_with_gaps_count = 0
        all_track_missing_frames = []
        clean_track_gap_frames = []
        worst_gap_track = None
        worst_clean_gap_track = None

    if EXPECT_PROCESSING_IN_EVERY_FRAME and len(failed_frames) > 0:
        final_result = "FAIL"
        final_reason = (
            "At least one checked frame had no accepted detected visual processing. "
            "For safety-critical visual processing verification, even one missed frame can be important."
        )
    elif clean_tracks_with_gaps_count > 0:
        final_result = "PASS_WITH_CLEAN_TRACK_GAP_WARNING"
        final_reason = (
            "Processing was detected overall, but clean temporal tracks still contain gaps. "
            "This may indicate missed visual processing in some frames."
        )
    elif frames_with_processing > 0:
        final_result = "PASS_PROCESSING_DETECTED"
        final_reason = (
            "Visual processing differences were detected in checked frames. "
            "False-positive filtering and track cleanup did not find critical clean-track gaps."
        )
    else:
        final_result = "NO_PROCESSING_DETECTED"
        final_reason = (
            "No accepted visual processing difference was detected between raw and processed videos."
        )

    if PROCESS_EVERY_N_FRAMES != 1 and final_result.startswith("PASS"):
        final_result = "PASS_WITH_SAMPLING_WARNING"
        final_reason += (
            " However, sampling was used, so unchecked frames may still contain missed processing."
        )

    summary_json = {
        "tool_version": OUTPUT_VERSION,
        "processed_video_file_name": PROCESSED_VIDEO_FILE_NAME,
        "raw_video_path": raw_video_path,
        "processed_video_path": processed_video_path,
        "output_directory": OUTPUT_DIR,
        "outputs": {
            "annotated_video": OUTPUT_ANNOTATED_VIDEO,
            "frame_report_csv": OUTPUT_FRAME_REPORT,
            "region_report_csv": OUTPUT_REGION_REPORT,
            "rejected_region_report_csv": OUTPUT_REJECTED_REGION_REPORT,
            "track_report_csv": OUTPUT_TRACK_REPORT,
            "summary_txt": OUTPUT_SUMMARY_TXT,
            "summary_json": OUTPUT_SUMMARY_JSON,
            "raw_frames_dir": RAW_FRAMES_DIR,
            "processed_frames_dir": PROCESSED_FRAMES_DIR,
            "annotated_frames_dir": ANNOTATED_FRAMES_DIR,
            "region_crops_dir": REGION_CROPS_DIR,
            "rejected_region_crops_dir": REJECTED_REGION_CROPS_DIR
        },
        "video_metadata": {
            "raw": raw_info,
            "processed": processed_info,
            "resolution_used": {
                "width": int(raw_width),
                "height": int(raw_height)
            },
            "total_common_frames": int(total_common_frames)
        },
        "settings": {
            "diff_threshold": DIFF_THRESHOLD,
            "min_box_area": MIN_BOX_AREA,
            "min_box_width": MIN_BOX_WIDTH,
            "min_box_height": MIN_BOX_HEIGHT,
            "merge_kernel_size": MERGE_KERNEL_SIZE,
            "box_padding": BOX_PADDING,
            "process_every_n_frames": PROCESS_EVERY_N_FRAMES,
            "max_frames": MAX_FRAMES,
            "expect_processing_in_every_frame": EXPECT_PROCESSING_IN_EVERY_FRAME,
