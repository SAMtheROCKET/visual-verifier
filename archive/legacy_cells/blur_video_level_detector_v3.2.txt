def normalize_bool_series(series):
    return (
        series
        .astype(str)
        .str.strip()
        .str.lower()
        .isin(["true", "1", "yes", "y"])

#================================================================
# OVERRIDE V3.0 TARGET LOADER FOR V3.2
#================================================================
# This patch makes V3.2 respect:
#   - is_active == True
#   - manual_review_decision != IGNORE
#
# It replaces the V3.0 load_or_build_target_regions() behavior only
# for this notebook runtime.

def load_or_build_target_regions(
    df_region_report,
    df_track_report,
    frame_width,
    frame_height
):
    if TARGET_REGION_SOURCE_MODE != "MANUAL_CSV":
        raise ValueError(
            "V3.2 expects TARGET_REGION_SOURCE_MODE = 'MANUAL_CSV'."
        )

    if not os.path.exists(MANUAL_TARGET_REGIONS_CSV_PATH):
        raise FileNotFoundError(
            f"Manual target CSV not found:\n{MANUAL_TARGET_REGIONS_CSV_PATH}"
        )

    target_df = pd.read_csv(MANUAL_TARGET_REGIONS_CSV_PATH)

    required_columns = [
        "frame_number",
        "target_id",
        "target_label",
        "x1",
        "y1",
        "x2",
        "y2"
    ]

    for column in required_columns:
        if column not in target_df.columns:
            raise ValueError(
                f"Manual target CSV is missing required column: {column}"
            )

    if "is_active" in target_df.columns:
        active_mask = normalize_bool_series(target_df["is_active"])
    else:
        active_mask = pd.Series([True] * len(target_df))

    if "manual_review_decision" in target_df.columns:
        keep_mask = (
            target_df["manual_review_decision"]
            .astype(str)
            .str.strip()
            .str.upper()
            .ne("IGNORE")
        )
    else:
        keep_mask = pd.Series([True] * len(target_df))

    target_df = target_df[active_mask & keep_mask].copy()

    if "source" not in target_df.columns:
        target_df["source"] = "MANUAL_CSV"
    else:
        target_df["source"] = target_df["source"].astype(str) + "+MANUAL_CSV_VALIDATION"

    target_df["v3_2_target_source_mode"] = "MANUAL_CSV"
    target_df["v3_2_used_for_validation"] = True

    return target_df, None, "MANUAL_CSV_LOADED_AND_FILTERED"
