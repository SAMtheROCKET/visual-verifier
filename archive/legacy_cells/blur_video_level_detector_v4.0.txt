def normalize_bool_value(value):
    return str(value).strip().lower() in ["true", "1", "yes", "y"]


def normalize_bool_series(series):
    return (
        series
        .astype(str)
        .str.strip()
        .str.lower()
        .isin(["true", "1", "yes", "y"])
    )


def safe_value(obj, default=""):
    if obj is None:
        return default

    try:
        if pd.isna(obj):
            return default
    except Exception:
        pass

    return obj


def sanitize_for_folder_name(value):
    value = str(value)
    invalid_chars = ['\\', '/', ':', '*', '?', '"', '<', '>', '|', ' ']

    for char in invalid_chars:
        value = value.replace(char, "_")

    return value


def convert_for_json_batch(obj):
    if obj is None:
        return None

    if isinstance(obj, dict):
        return {str(k): convert_for_json_batch(v) for k, v in obj.items()}

    if isinstance(obj, list):
        return [convert_for_json_batch(v) for v in obj]

    if isinstance(obj, tuple):
        return [convert_for_json_batch(v) for v in obj]

    if hasattr(obj, "item"):
        try:
            return obj.item()
        except Exception:
            pass

    try:
        if pd.isna(obj):
            return None
    except Exception:
        pass

    return obj


def extract_summary_field(summary, section, key, default=None):
    try:
        return summary.get(section, {}).get(key, default)
    except Exception:
        return default

# BATCH TARGET LOADER OVERRIDE
#================================================================
# This supports both:
#   MANUAL_CSV
#   AUTO_BOOTSTRAP_FROM_BEST_TRACK
#
# It replaces the target loader in notebook runtime for V4.0 batch use.

def load_or_build_target_regions(
    df_region_report,
    df_track_report,
    frame_width,
    frame_height
):
    global TARGET_REGION_SOURCE_MODE
    global MANUAL_TARGET_REGIONS_CSV_PATH

    if TARGET_REGION_SOURCE_MODE == "MANUAL_CSV":
        if not os.path.exists(MANUAL_TARGET_REGIONS_CSV_PATH):
            raise FileNotFoundError(
                f"Manual target CSV not found:\n{MANUAL_TARGET_REGIONS_CSV_PATH}"
            )

        target_df = pd.read_csv(MANUAL_TARGET_REGIONS_CSV_PATH)

        required_columns = [
            "frame_number",
            "target_id",
            "target_label",
            "x1",
            "y1",
            "x2",
            "y2"
        ]

        for column in required_columns:
            if column not in target_df.columns:
                raise ValueError(
                    f"Manual target CSV is missing required column: {column}"
                )

        if "is_active" in target_df.columns:
            active_mask = normalize_bool_series(target_df["is_active"])
        else:
            active_mask = pd.Series([True] * len(target_df))

        if "manual_review_decision" in target_df.columns:
            keep_mask = (
                target_df["manual_review_decision"]
                .astype(str)
                .str.strip()
                .str.upper()
                .ne("IGNORE")
            )
        else:
            keep_mask = pd.Series([True] * len(target_df))

        target_df = target_df[active_mask & keep_mask].copy()

        if "source" not in target_df.columns:
            target_df["source"] = "MANUAL_CSV"
        else:
            target_df["source"] = target_df["source"].astype(str) + "+MANUAL_CSV_BATCH_VALIDATION"

        target_df["v4_0_target_source_mode"] = "MANUAL_CSV"
        target_df["v4_0_used_for_batch_validation"] = True

        return target_df, None, "MANUAL_CSV_LOADED_AND_FILTERED"

    if TARGET_REGION_SOURCE_MODE == "AUTO_BOOTSTRAP_FROM_BEST_TRACK":
        target_df, selected_track_id = build_auto_target_regions_from_track(
            df_region_report=df_region_report,
            df_track_report=df_track_report,
            frame_width=frame_width,
            frame_height=frame_height
        )

        return target_df, selected_track_id, "AUTO_BOOTSTRAP_FROM_BEST_TRACK"

    raise ValueError(
        f"Unsupported TARGET_REGION_SOURCE_MODE: {TARGET_REGION_SOURCE_MODE}"
    )

    global MANUAL_TARGET_REGIONS_CSV_PATH

        BATCH_OUTPUT_ROOT,

    OUTPUT_ANNOTATED_VIDEO = os.path.join(
        OUTPUT_DIR,
        f"{PROCESSED_VIDEO_FILE_NAME}_{OUTPUT_VERSION}_annotated_video.mp4"

        OUTPUT_DIR,
        f"{PROCESSED_VIDEO_FILE_NAME}_{OUTPUT_VERSION}_frame_report.csv"

        OUTPUT_DIR,
        f"{PROCESSED_VIDEO_FILE_NAME}_{OUTPUT_VERSION}_region_report.csv"
    )

    OUTPUT_REJECTED_REGION_REPORT = os.path.join(
        OUTPUT_DIR,
        f"{PROCESSED_VIDEO_FILE_NAME}_{OUTPUT_VERSION}_rejected_region_report.csv"
    )

        OUTPUT_DIR,

        OUTPUT_DIR,

    )

        OUTPUT_DIR,

        f"{PROCESSED_VIDEO_FILE_NAME}_{OUTPUT_VERSION}_summary.json"

    PROCESSED_FRAMES_DIR = os.path.join(OUTPUT_DIR, "processed_frames")
    ANNOTATED_FRAMES_DIR = os.path.join(OUTPUT_DIR, "annotated_frames")

    REJECTED_REGION_CROPS_DIR = os.path.join(OUTPUT_DIR, "rejected_region_crops")
    TARGET_CROPS_DIR = os.path.join(OUTPUT_DIR, "target_crops")

    os.makedirs(REJECTED_REGION_CROPS_DIR, exist_ok=True)

    )

        "case_id": case_id,
        "processed_video_file_name": processed_name,
        "output_dir": OUTPUT_DIR
    }

        "processed_video_file_name",
        "raw_video_path",
        "processed_video_path",
        "target_region_source_mode",
        "manual_target_regions_csv_path"
    ]

    for column in required_columns:
        if column not in batch_cases_df.columns:
            raise ValueError(
                f"Batch cases CSV is missing required column: {column}"
            )

    enabled_mask = batch_cases_df["enabled"].apply(normalize_bool_value)
    enabled_cases_df = batch_cases_df[enabled_mask].copy()

    batch_rows = []
    batch_log_lines = []

    batch_started_at = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    print("====================================================")
    print("V4.0 BATCH PROCESSING STARTED")
    print("====================================================")
    print(f"Batch cases CSV: {BATCH_CASES_CSV}")
    print(f"Enabled cases: {len(enabled_cases_df)}")
    print(f"Batch output root: {BATCH_OUTPUT_ROOT}")

    batch_log_lines.append("====================================================")
    batch_log_lines.append("V4.0 BATCH PROCESSING LOG")
    batch_log_lines.append("====================================================")
    batch_log_lines.append(f"Started at: {batch_started_at}")
    batch_log_lines.append(f"Batch cases CSV: {BATCH_CASES_CSV}")
    batch_log_lines.append(f"Enabled cases: {len(enabled_cases_df)}")
    batch_log_lines.append("")

    for batch_index, (_, case_row) in enumerate(enabled_cases_df.iterrows(), start=1):
        case_id = str(case_row["case_id"])

        print("\n====================================================")
        print(f"RUNNING CASE {batch_index}/{len(enabled_cases_df)}: {case_id}")
        print("====================================================")

        case_info = configure_case_globals(case_row)

        case_result = {
            "case_id": case_id,
            "processed_video_file_name": case_info["processed_video_file_name"],
            "raw_video_path": RAW_VIDEO_PATH,
            "processed_video_path": PROCESSED_VIDEO_PATH,
            "target_region_source_mode": TARGET_REGION_SOURCE_MODE,
            "manual_target_regions_csv_path": MANUAL_TARGET_REGIONS_CSV_PATH,
            "output_dir": OUTPUT_DIR,
            "case_status": "UNKNOWN",
            "error_message": "",
            "final_result": "",
            "final_reason": "",
            "frames_checked": 0,
            "frames_with_detected_processing": 0,
            "frames_without_detected_processing": 0,
            "processing_coverage_percent": 0.0,
            "failed_frame_count": 0,
            "failed_frames": [],
            "total_targets_checked": 0,
            "targets_covered_count": 0,
            "targets_failed_count": 0,
            "target_coverage_percent": 0.0,
            "target_failed_frames": [],
            "total_candidate_regions": 0,
            "total_accepted_regions": 0,
            "total_rejected_regions": 0,
            "unique_tracks_count": 0,
            "clean_tracks_with_gaps_count": 0,
            "clean_track_gap_frames": [],
            "summary_json": OUTPUT_SUMMARY_JSON,
            "summary_txt": OUTPUT_SUMMARY_TXT,
            "annotated_video": OUTPUT_ANNOTATED_VIDEO,
            "frame_report_csv": OUTPUT_FRAME_REPORT,
            "region_report_csv": OUTPUT_REGION_REPORT,
            "track_report_csv": OUTPUT_TRACK_REPORT,
            "target_validation_report_csv": OUTPUT_TARGET_VALIDATION_REPORT
        }

        try:
            if not os.path.exists(RAW_VIDEO_PATH):
                raise FileNotFoundError(f"Raw video not found: {RAW_VIDEO_PATH}")

            if not os.path.exists(PROCESSED_VIDEO_PATH):
                raise FileNotFoundError(f"Processed video not found: {PROCESSED_VIDEO_PATH}")

            if TARGET_REGION_SOURCE_MODE == "MANUAL_CSV":
                if not os.path.exists(MANUAL_TARGET_REGIONS_CSV_PATH):
                    raise FileNotFoundError(
                        f"Manual target CSV not found: {MANUAL_TARGET_REGIONS_CSV_PATH}"
                    )

            (
                df_frame_report,
                df_region_report,
                df_rejected_region_report,
                df_track_report,
                df_target_regions,
                df_target_validation_report,
                case_summary
            ) = verify_video_level_v3_0(
                RAW_VIDEO_PATH,
                PROCESSED_VIDEO_PATH

            case_result["targets_covered_count"] = target_summary.get("targets_covered_count", 0)
            case_result["targets_failed_count"] = target_summary.get("targets_failed_count", 0)
            case_result["target_coverage_percent"] = target_summary.get("target_coverage_percent", 0.0)
            case_result["target_failed_frames"] = target_summary.get("target_failed_frames", [])

            case_result["total_candidate_regions"] = fp_summary.get("total_candidate_regions", 0)
            case_result["total_accepted_regions"] = fp_summary.get("total_accepted_regions", 0)

    print("")

    print("")

    print("")

    print("BATCH SUMMARY PREVIEW")
    print("====================================================")

        "target_failed_frames",
        "output_dir"
    ]

    existing_preview_columns = [
        column for column in preview_columns
        if column in batch_summary_df.columns
    ]

    print(batch_summary_df[existing_preview_columns])

    return batch_summary_df, batch_summary_json
